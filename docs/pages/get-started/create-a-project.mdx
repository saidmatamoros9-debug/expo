import React, { useEffect, useState } from "react";
import "./SplashScreen.css";

export default function SplashScreen({ onFinish }) {
  const [loadingText, setLoadingText] = useState("Inicializando sistema...");

  useEffect(() => {
    // 🔊 Sonido de inicio
    const audio = new Audio("/jarvis-startup.mp3");
    audio.volume = 0.7;
    audio.play();

    const steps = [
      "Cargando módulos...",
      "Verificando integridad...",
      "Conectando a J.A.R.V.I.S...",
      "Sistema operativo online...",
      "Listo."
    ];

    let index = 0;
    const interval = setInterval(() => {
      setLoadingText(steps[index]);
      index++;

      if (index === steps.length) {
        clearInterval(interval);
        setTimeout(onFinish, 800);
      }
    }, 900);

    return () => clearInterval(interval);
  }, [onFinish]);

  return (
    <div className="splash-container">
      <h1 className="jarvis-title">JARVIS</h1>
      <p className="subtitle">ASISTENTE VIRTUAL</p>

      <div className="loader"></div>

      <p className="loading-text">{loadingText}</p>
    </div>
  );
}

import React, { useState } from "react";
import SplashScreen from "./SplashScreen";
import "./App.css";

export default function App() {
  const [showSplash, setShowSplash] = useState(true);
  const [message, setMessage] = useState("");
  const [chat, setChat] = useState([]);

  const finishSplash = () => setShowSplash(false);

  const speak = (text) => {
    const speech = new SpeechSynthesisUtterance(text);
    speech.lang = "es-ES";
    window.speechSynthesis.speak(speech);
  };

  const sendMessage = () => {
    if (!message) return;

    const userMessage = { sender: "Usuario", text: message };
    setChat((prev) => [...prev, userMessage]);

    let responseText = "No entendí tu mensaje";

    if (message.toLowerCase().includes("hola")) {
      responseText = "Hola señor";
    }

    if (message.toLowerCase().includes("hora")) {
      responseText = `La hora actual es ${new Date().toLocaleTimeString()}`;
    }

    const botMessage = { sender: "Jarvis", text: responseText };
    setChat((prev) => [...prev, botMessage]);

    // 🔊 Sonido al responder
    const beep = new Audio("/jarvis-beep.mp3");
    beep.volume = 0.5;
    beep.play();

    // 🔊 Voz de Jarvis
    speak(responseText);

    setMessage("");
  };

  const startVoiceRecognition = () => {
    const recognition = new window.webkitSpeechRecognition();
    recognition.lang = "es-ES";

    recognition.onresult = (event) => {
      const voiceText = event.results[0][0].transcript;
      setMessage(voiceText);
    };

    recognition.start();
  };

  if (showSplash) return <SplashScreen onFinish={finishSplash} />;

  return (
    <div className="container">
      <h1>Jarvis</h1>

      <div className="chat-box">
        {chat.map((msg, index) => (
          <div key={index} className="message">
            <strong>{msg.sender}: </strong>
            {msg.text}
          </div>
        ))}
      </div>

      <div className="input-area">
        <input
          type="text"
          placeholder="Escribe un mensaje..."
          value={message}
          onChange={(e) => setMessage(e.target.value)}
        />

        <button onClick={sendMessage}>Enviar</button>
        <button onClick={startVoiceRecognition}>🎤</button>
      </div>
    </div>
  );
}
